// كود زر إرسال المنتج لواجهة المحل الخاصة (المربع التحتاني) function sendToStoreFront(index) { let storage = JSON.parse(localStorage.getItem('vendorStorage')) || []; let storeFront = JSON.parse(localStorage.getItem('storeFrontDisplayOnly')) || []; // نقل المنتج المحدد لواجهة المحل فقط storeFront = [storage[index]]; localStorage.setItem('storeFrontDisplayOnly', JSON.stringify(storeFront)); alert('تم النشر في واجهة المحل الخاصة بنجاح!'); loadDashboard(); } // كود زر إرسال المنتج للسلايدر الرئيسي (الشاشة الرئيسية فوق) function sendToMainScreen(index) { let storage = JSON.parse(localStorage.getItem('vendorStorage')) || []; let mainScreen = JSON.parse(localStorage.getItem('mainSliderProducts')) || []; mainScreen.push(storage[index]); localStorage.setItem('mainSliderProducts', JSON.stringify(mainScreen)); alert('تم النشر في الشاشة الرئيسية للمول بنجاح!'); loadDashboard(); }
